\ncdsubsec{二、Dialog \& Udtale（课文对话、发音标注与中文译文）}

\begin{tipbox}{场景说明}
艾玛（Emma）在哥本哈根看上一套新挂牌出租的公寓，房东亨里克（Henrik）带她实地看房。艾玛仔细询问了面积、房间配置、楼层、月租以及周边超市与交通情况。
\end{tipbox}

\dialogturn{1}{Emma}{Hej Henrik! Tak fordi vi må se den nye lejlighed i dag.}
{\textit{fordi} \ipa{fɒˈdiːˀ}（末尾 d 不发音）；\textit{den nye} \ipa{dɛn ˈnyːə}（定指形容词加 \textit{-e}，发清晰的央元音 \ipa{ə}）；\textit{lejlighed} \ipa{ˈlɑɪ̯liˌheðˀ}（中间 \textit{-ig-} 中的 g 不发音，词尾软 d 带轻微喉塞音）。}
{嗨，亨里克！谢谢你今天让我们来看这套新公寓。}

\dialogturn{2}{Henrik}{Velkommen indenfor! Ja, det er en lys og dejlig lejlighed på anden sal.}
{\textit{indenfor} \ipa{ˈenənˌfɒː}（\textit{-nd-} 中的 \textbf{d 不发音}）；\textit{lys} \ipa{ˈlyːˀs}（长圆唇音 \ipa{y} + 喉塞音）；\textit{dejlig} \ipa{ˈdɑɪ̯li}（词尾 \textit{-ig} 中 g 不发音）；\textit{på anden sal} \ipa{pʰɔ ˈanən ˈsæːˀl}（注意丹麦语“二楼 \textit{anden sal}”等于中国的三楼！）。}
{欢迎进来！是的，这是一套明亮又舒适的公寓，在（丹麦的）二楼（即地面三层）。}

\dialogturn{3}{Emma}{Hvor stor er lejligheden, og hvor mange værelser har den?}
{\textit{stor} \ipa{ˈsdoɐ̯ˀ}（o 偏向 \ipa{o}，r 弱化成 \ipa{ɐ̯}）；\textit{lejligheden} \ipa{ˈlɑɪ̯liˌheðˀən}；\textit{værelser} \ipa{ˈvɛːɐlsɐ}（词首 v，中间 \textit{-el-} 轻弱，词尾 \textit{-er} 读 \ipa{ɐ}）。}
{这套公寓有多大，有几个房间呢？}

\dialogturn{4}{Henrik}{Den er 65 kvadratmeter. Der er en stor stue, et lille soveværelse, et nyt køkken og et gammelt badeværelse.}
{\textit{65} \ipa{ˈfɛmˀɔˌtʰʁɛs}（65 读作 \textit{femogtres}）；\textit{kvadratmeter} \ipa{kʰvaˈdʁɑːˀdˌmeːˀdɐ}；\textit{nyt} \ipa{ˈnyd}（短元音，没有喉塞音）；\textit{køkken} \ipa{ˈkʰøɡən}；\textit{gammelt} \ipa{ˈɡaməld}（末尾 l 清晰，t 轻读接近 d）。}
{面积是 65 平方米。里面有一间大客厅、一间小卧室、一个新厨房和一间老浴室。}

\dialogturn{5}{Emma}{Hvor er stuen lys! Store vinduer er virkelig dejlige. Har lejligheden også en altan?}
{\textit{stuen} \ipa{ˈsduːən}；\textit{store} \ipa{ˈsdoːɐ}（加了 \textit{-e} 后长音不变，r 读 \ipa{ɐ}）；\textit{vinduer} \ipa{ˈvenduɐ}（\textit{-nd-} 中 d 不发音）；\textit{altan} \ipa{alˈtʰæːˀn}（重音在第二音节）。}
{客厅真亮堂啊！大窗户真是太棒了。这套公寓带阳台吗？}

\dialogturn{6}{Henrik}{Ja, den har en lille altan ud mod en rolig gård. De gamle huse her i gaden er meget stille.}
{\textit{rolig} \ipa{ˈʁoːli}（词尾 g 不发音）；\textit{gård} \ipa{ˈɡɒːˀ}（\textit{-rd} 中的 d 不发音）；\textit{gamle} \ipa{ˈɡamlə}（注意 \textit{gammel} 变复数/定指时少了一个 m 和 e）；\textit{stille} \ipa{ˈsdilə}（形容词不变形）。}
{带的，它有一个朝向安静庭院的小阳台。这条街上的老房子都非常安静。}

\dialogturn{7}{Emma}{Det lyder godt. Hvor meget koster huslejen om måneden? Er den billig?}
{\textit{lyder} \ipa{ˈlyːðˀɐ}（软 d）；\textit{huslejen} \ipa{ˈhuːsˌlɑɪ̯ən}；\textit{om måneden} \ipa{ʌm ˈmɔːnəðən}（词尾软 d 极轻）；\textit{billig} \ipa{ˈbili}（末尾 g 不发音）。}
{听起来真不错。每月房租多少钱？算便宜吗？}

\dialogturn{8}{Henrik}{Den er ret billig for København: 7.500 kroner om måneden plus varme og vand.}
{\textit{7.500} \ipa{ˈsyʊ̯ˀ ˈtʰuːˀsən ˈfɛmˀ ˈhunəðə}（\textit{syv tusind fem hundrede}）；\textit{varme} \ipa{ˈvɑːmə}；\textit{vand} \ipa{ˈvanˀ}（\textit{-nd} 中的 \textbf{d 不发音}，带喉塞音）。}
{在哥本哈根算挺便宜的：每月 7500 克朗，另外加暖气费和水费。}

\dialogturn{9}{Emma}{Hvordan er området? Er der et supermarked og offentlig transport tæt på?}
{\textit{området} \ipa{ˈʌmˌʁɔːðəð}（软 d）；\textit{offentlig} \ipa{ˈʌfənli}（\textbf{t 和末尾 g 都不发音}！）；\textit{tæt på} \ipa{ˈtʰɛd ˈpʰɔːˀ}。}
{周边环境怎么样？附近有超市和公共交通吗？}

\dialogturn{10}{Henrik}{Ja, der ligger to billige supermarkeder lige om hjørnet, en grøn park og metroen kun fem minutter herfra.}
{\textit{billige} \ipa{ˈbiliə}；\textit{hjørnet} \ipa{ˈjɶɐ̯nəð}（词首 \textbf{\textit{hj-} 中 h 不发音}，词尾 \textit{-et} 读 \ipa{əð}）；\textit{minutter} \ipa{miˈnudɐ}（重音在第二音节）。}
{有的，拐角处就有两家平价超市，还有一座绿地公园，地铁站离这儿只有五分钟路程。}

\ncdsubsec{四、Grammatik \& Faste vendinger（核心语法与地道表达精讲）}

\begin{grammatikbox}{语法点 1：形容词基础词尾变化（Grundformer）}
丹麦语形容词无论放在\textbf{名词前面}（作定语，如 \textit{en stor stue}）还是放在\textbf{动词 \textit{er} 后面}（作表语，如 \textit{Stuen er stor}），都必须严格根据它修饰的名词的\textbf{性（\textit{en/et}）、数（单/复）和定指性}来变化词尾：

\vspace{0.3em}
\renewcommand{\arraystretch}{1.25}
{\small
\begin{tabularx}{\linewidth}{@{} >{\bfseries\raggedright\arraybackslash}p{3.2cm} X X X @{}}
\toprule
\textbf{词组类型} & \textbf{变化规则} & \textbf{名词前（定语）} & \textbf{系动词后（表语）} \\
\midrule
不定指 \textbf{\textit{en}-词} 单数 & \textbf{原形}（不加词尾） & \textit{en \da{stor} stue} & \textit{Stuen er \da{stor}.} \\
不定指 \textbf{\textit{et}-词} 单数 & \textbf{加 \dabold{-t}} & \textit{et \dabold{stort} køkken} & \textit{Køkkenet er \dabold{stort}.} \\
\textbf{复数}（无论 \textit{en/et}） & \textbf{加 \dabold{-e}} & \textit{to \dabold{store} vinduer} & \textit{Vinduerne er \dabold{store}.} \\
\textbf{定指形式}（见语法点 3） & \textbf{加 \dabold{-e}} & \textit{den \dabold{store} stue} & \textit{Den er \dabold{stor}.}（指代单数） \\
\bottomrule
\end{tabularx}
}
\vspace{0.3em}

\textbf{特别提醒}：在作表语时，看主语本身决定词尾：\textit{Lejligheden (\textit{en}-词) er \da{ny}.} $\rightarrow$ \textit{Køkkenet (\textit{et}-词) er \dabold{nyt}.} $\rightarrow$ \textit{Værelserne (复数) er \dabold{nye}.}
\end{grammatikbox}

\begin{grammatikbox}{语法点 2：高频特殊与不变形形容词一览}
并非所有形容词都机械地加 \textit{-t} 和 \textit{-e}。以下 4 类高频变化是 DU3 Modul 1--2 的核心考点，必须整组背熟：

\vspace{0.3em}
\renewcommand{\arraystretch}{1.25}
{\small
\begin{tabularx}{\linewidth}{@{} >{\raggedright\arraybackslash}p{3.6cm} >{\itshape\raggedright\arraybackslash}p{2.3cm} >{\itshape\raggedright\arraybackslash}p{2.5cm} >{\itshape\raggedright\arraybackslash}p{2.5cm} X @{}}
\toprule
\textbf{变化规律} & \textbf{\textit{en}-词单数} & \textbf{\textit{et}-词单数} & \textbf{复数 / 定指} & \textbf{典型例词} \\
\midrule
\ding{172} \textbf{\textit{lille / små} 完全异干} & lille & lille（\textbf{不加 t}） & \dabold{små} & \textit{et lille køkken} / \textit{to små værelser} \\
\ding{173} \textbf{\textit{-el / -en / -er} 缩音} & gammel & gammelt & \dabold{gamle} & 复数吞掉中间元音且单写 m \\
\ding{174} \textbf{单音节长元音 / 双辅音} & grøn\newline ny & grøn\dabold{t}\newline ny\dabold{t} & grøn\dabold{ne}\newline ny\dabold{e} & 短元音词尾单辅音加 \textit{-e} 时双写（\textit{grønne}） \\
\ding{175} \textbf{以 \textit{-e} 结尾}（不变形） & stille\newline moderne & stille\newline moderne & stille\newline moderne & 词尾已是 \textit{-e}，永远不变形！（还有 \textit{gratis}） \\
\bottomrule
\end{tabularx}
}
\vspace{0.3em}

\textbf{小补充}：以 \textit{-sk} 结尾的形容词（如 \textit{dansk, engelsk, praktisk}）在 \textit{et}-词后也\textbf{不加 \textit{-t}}（如 \textit{et dansk ord}），复数加 \textit{-e}（\textit{danske ord}）。
\end{grammatikbox}

\begin{grammatikbox}{语法点 3：定指结构“\textit{den / det / de} + 形容词\textit{-e} + 名词原形”与固定搭配}
当“the + 形容词 + 名词”一起出现时，丹麦语\textbf{不能}把定冠词后缀留在名词末尾，而必须用独立前冠词 \dabold{den / det / de} 领衔：
\begin{itemize}
  \item \textbf{黄金公式}：\da{den}（共性单数）/ \da{det}（中性单数）/ \da{de}（复数） + \textbf{形容词-\dabold{e}} + \textbf{名词原形（不加 \textit{-en/-et}！）}
  \item 正确：\textit{\da{den nye} lejlighed}（这套新公寓）/ \textit{\da{det lille} soveværelse}（那间小卧室）/ \textit{\da{de gamle} huse}（那些老房子）
  \item 绝对错误：\textit{*den nye lejlighed\dabold{en}}（定冠词前呼后应重叠是考场头号扣分错误！）
\end{itemize}

\textbf{高频地道搭配（Faste vendinger）：}
\begin{itemize}
  \item \da{en lejlighed til leje} —— 待出租的公寓。（同款：\textit{et værelse til leje} 出租房间）
  \item \da{på første / anden / tredje sal} —— 在（丹麦的）一楼 / 二楼 / 三楼。（注意：丹麦的底层叫 \textit{stuen}，所以 \textit{første sal} 相当于中国的二楼，\textit{anden sal} 相当于中国的三楼！）
  \item \da{lige om hjørnet} / \da{tæt på} —— 就在拐角处 / 离……很近。
  \item \da{ud mod gården / gaden} —— 朝向庭院 / 朝向街道。
\end{itemize}
\end{grammatikbox}

\ncdsubsec{五、Mønsterøvelser（新概念式句型操练）}

\textbf{练习 A：用括号内形容词的正确形式填空（注意 \textit{en/et}、复数与定指）}
\begin{enumerate}
  \item Lejligheden har en \underline{\hspace{2.2cm}} stue og et \underline{\hspace{2.2cm}} køkken. (\textit{stor}, \textit{ny})
  \item Mit soveværelse er meget \underline{\hspace{2.2cm}}, men det er \underline{\hspace{2.2cm}}. (\textit{lille}, \textit{rolig})
  \item Husene i denne gade er ret \underline{\hspace{2.2cm}}, men vinduerne er \underline{\hspace{2.2cm}}. (\textit{gammel}, \textit{stor})
  \item Vi vil gerne se \underline{\hspace{2.4cm}} altan ud mod gården. (\textit{the + lille})
  \item Der ligger to \underline{\hspace{2.4cm}} supermarkeder lige om hjørnet. (\textit{billig})
  \item Parkerne i København er altid \underline{\hspace{2.2cm}} om foråret. (\textit{grøn})
  \item Kan du lide \underline{\hspace{2.8cm}} badeværelse? (\textit{the + ny})
  \item Området er meget \underline{\hspace{2.2cm}} om aftenen. (\textit{stille})
\end{enumerate}

\textbf{练习 B：按示范将不定指词组改写为定指词组（注意去掉名词后缀并调整形容词！）}
\begin{quote}
\small 例：\textit{en ny lejlighed} $\rightarrow$ \textit{\textbf{den nye} lejlighed} \qquad \textit{et gammelt hus} $\rightarrow$ \textit{\textbf{det gamle} hus}
\end{quote}
\begin{enumerate}
  \item \textit{en stor stue} $\rightarrow$ \underline{\hspace{4.5cm}}
  \item \textit{et lyst værelse} $\rightarrow$ \underline{\hspace{4.5cm}}
  \item \textit{et lille køkken} $\rightarrow$ \underline{\hspace{4.5cm}}
  \item \textit{to små soveværelser} $\rightarrow$ \underline{\hspace{4.5cm}}（改写为“那些小卧室”）
  \item \textit{et billigt supermarked} $\rightarrow$ \underline{\hspace{4.5cm}}
\end{enumerate}

\textbf{练习 C：中译丹实战}
\begin{enumerate}
  \item 这套明亮的新公寓在二楼（按丹麦楼层说法），有 65 平方米。
  \item 公寓里有一间大客厅、一个新厨房和两间小卧室。
  \item 拐角处有一家便宜的超市和一个安静的小公园。
  \item 这套公寓每月房租多少钱？（用 \textit{om måneden}）
  \item 那些老房子很安静，但我更喜欢这套新公寓。
\end{enumerate}

\ncdsubsec{六、Resumé \& Mundtlig træning（口语复述与参考答案）}

\textbf{口语复述任务（对标 Modul 1--2 住房口试表达）}：请合上课文，先用丹麦语回答 \textbf{Før du læser} 的两个问题，然后尝试用 \textbf{4--5 句话} 介绍艾玛看的这套公寓（面积、房间、楼层、租金和周边环境）。

\begin{answerbox}
\textbf{练习 A 答案}
\begin{enumerate}
  \item \da{stor}（\textit{en stue} 为共性词，原形）；\da{nyt}（\textit{et køkken} 为中性词，加 \textit{-t}）
  \item \da{lille}（\textit{lille} 在中性词后不加 \textit{-t}）；\da{roligt}（\textit{det = soveværelse} 为中性词，加 \textit{-t}）
  \item \da{gamle}（主语 \textit{Husene} 是复数）；\da{store}（主语 \textit{vinduerne} 是复数）
  \item \da{den lille}（\textit{en altan} 定指单数用 \textit{den} + \textit{lille}）
  \item \da{billige}（复数名词前加 \textit{-e}）
  \item \da{grønne}（复数主语 \textit{Parkerne}，单辅音 n 双写再加 \textit{-e}）
  \item \da{det nye}（\textit{et badeværelse} 定指用 \textit{det} + \textit{nye}）
  \item \da{stille}（以 \textit{-e} 结尾，在中性主语 \textit{Området} 后仍不变形）
\end{enumerate}

\textbf{练习 B 答案}
\begin{enumerate}
  \item \da{den store stue}
  \item \da{det lyse værelse}
  \item \da{det lille køkken}
  \item \da{de små soveværelser}（复数定指用 \textit{de}，\textit{lille} 变 \textit{små}，名词还原为不定指复数 \textit{soveværelser}）
  \item \da{det billige supermarked}
\end{enumerate}

\textbf{练习 C 答案}
\begin{enumerate}
  \item \textit{Den nye, lyse lejlighed ligger på anden sal og er 65 kvadratmeter.}
  \item \textit{Der er en stor stue, et nyt køkken og to små soveværelser i lejligheden.}
  \item \textit{Der ligger et billigt supermarked og en stille, lille park lige om hjørnet.}
  \item \textit{Hvor meget koster huslejen i lejligheden om måneden?}
  \item \textit{De gamle huse er meget stille, men jeg kan bedre lide den nye lejlighed.}
\end{enumerate}

\textbf{Før du læser 参考回答 \& 口语复述范文（Resumé）}

  \item \textbf{问题 2}：\textit{Huslejen koster 7.500 kroner om måneden plus varme og vand. Tæt på lejligheden ligger der to billige supermarkeder, en grøn park og metroen.}
  \item \textbf{口语复述范文}：\textit{Emma ser på en ny, lys lejlighed på anden sal i København. Den er 65 kvadratmeter stor og har en stor stue, et lille soveværelse, et nyt køkken og et gammelt badeværelse. Den har også en lille altan ud mod en rolig gård. Huslejen koster 7.500 kroner om måneden plus varme og vand, og både supermarkeder, en park og metroen ligger tæt på.}
