\ncdsubsec{二、Dialog \& Udtale（课文对话、发音标注与中文译文）}

\begin{tipbox}{场景说明}
卢卡斯（Lukas）第一天到一家丹麦手工面包店打工，店主梅特（Mette）带他换上工作服，交代厨房卫生规范、收银台服务礼仪以及手机管理规定。
\end{tipbox}

\dialogturn{1}{Mette}{Velkommen til din første dag, Lukas! Kom ind i køkkenet, og tag et rent forklæde på.}
{\textit{Kom} \ipa{ˈkʰʌmˀ}（祈使句触发喉塞音 \ipa{ˀ}）；\textit{tag} \ipa{ˈtʰæːˀ}（祈使句，末尾 \textbf{g 不发音}）；\textit{forklæde} \ipa{ˈfɒːˌklɛːðə}（软 d）。}
{欢迎来到你上班的第一天，卢卡斯！来厨房里面，穿上一件干净的围裙。}

\dialogturn{2}{Lukas}{Tak, Mette! Hvad skal man huske allerførst her i bageriet?}
{\textit{man} \ipa{man}（泛指代词轻读，\textbf{无喉塞音}，区别于 \textit{mand} \ipa{ˈmanˀ}）；\textit{allerførst} \ipa{ˈalɐˌfɶɐ̯sd}；\textit{bageriet} \ipa{bæːɐˈʁiːˀəð}。}
{谢谢，梅特！在这家面包店里，大家最先必须记住什么呢？}

\dialogturn{3}{Mette}{Vask altid hænderne grundigt først! Man skal have rent tøj på, og man må ikke have ringe eller ur på.}
{\textit{Vask} \ipa{ˈvasɡ}（短元音 + 清辅音，无喉塞音）；\textit{hænderne} \ipa{ˈhɛnˀɐnə}（\textit{-nd-} 的 d 不发音）；\textit{grundigt} \ipa{ˈɡʁɔnˀdi}（词尾 \textit{-gt} 不发音）。}
{永远要先彻底洗手！工作时必须穿干净的衣服，而且不准戴戒指或手表。}

\dialogturn{4}{Lukas}{Forstået! Behøver jeg så at bage brødet i dag?}
{\textit{Forstået} \ipa{fɒˈsdɔːˀð}（重音在 \textit{-stå-}，长音带喉塞音）；\textit{Behøver} \ipa{beˈhøːˀvɐ}；\textit{brødet} \ipa{ˈbʁœðˀəð}（软 d）。}
{明白了！那我今天需要烤面包吗？}

\dialogturn{5}{Mette}{Nej, du behøver ikke at bage endnu. Vær sød at hjælpe ved disken, og pak de friske rundstykker.}
{\textit{Vær} \ipa{ˈvɛɐ̯ˀ}（单音节祈使词，带喉塞音）；\textit{hjælpe} \ipa{ˈjɛlbə}（h 不发音）；\textit{pak} \ipa{ˈpʰaɡ}；\textit{rundstykker} \ipa{ˈʁɔnˌsdøɡɐ}（\textit{rund-} 中 d 不发音）。}
{不用，你还不需要烤面包。请帮忙照看一下柜台，把新鲜的小圆面包打包好。}

\dialogturn{6}{Lukas}{Selvfølgelig! Skal man bruge handsker, når man pakker brød og kager?}
{\textit{Selvfølgelig} \ipa{seˈfølˀəli}（注意 \textit{v} 和两个 \textit{g} 均不发音！）；\textit{handsker} \ipa{ˈhansgɐ}（\textit{-nd-} 的 d 不发音）。}
{没问题！大家打包面包和糕点的时候必须戴手套吗？}

\dialogturn{7}{Mette}{Ja, brug altid en tang eller handsker. Og husk at smile venligt til kunderne!}
{\textit{brug} \ipa{ˈbʁuːˀ}（祈使句，\textbf{g 静音}且带喉塞音！）；\textit{tang} \ipa{ˈtʰaŋˀ}；\textit{venligt} \ipa{ˈvɛnli}（\textit{-igt} 的 g 和 t 均不发音）。}
{是的，永远要用面包夹或手套。还有，记得面带微笑友善地面对顾客！}

\dialogturn{8}{Lukas}{Intet problem! Må man have sin mobiltelefon i lommen?}
{\textit{mobiltelefon} \ipa{moˈbiːˀlˌtʰeləfoːˀn}；\textit{lommen} \ipa{ˈlʌmən}（双写 m 读单一清晰鼻音）。}
{没问题！大家可以把手机放在口袋里吗？}

\dialogturn{9}{Mette}{Nej, man må ikke bruge mobilen ved disken. Vær sød at slukke den og lægge den i skabet! Skynd dig lidt, vi åbner om fem minutter.}
{\textit{slukke} \ipa{ˈslɔɡə}；\textit{lægge} \ipa{ˈlɛɡə}；\textit{Skynd dig} \ipa{ˈsgønˀ dɑɪ̯}（\textit{-nd} 的 d 不发音，带喉塞音）。}
{不行，不可以在柜台边使用手机。请把它关掉放进柜子里！动作稍微快一点，我们五分钟后就开门营业了。}

\dialogturn{10}{Lukas}{Helt i orden! Jeg slukker den nu og kommer ud med det samme.}
{\textit{Helt i orden} \ipa{ˈheːˀld i ˈɒːˀdən}；\textit{med det samme} \ipa{mɛ de ˈsamə}（日常口语中 \textit{det} 轻短弱读）。}
{完全没问题！我现在就关机，马上出来。}

\ncdsubsec{四、Grammatik \& Faste vendinger（核心语法与地道表达精讲）}

\begin{grammatikbox}{语法点 1：祈使句（Imperativ）构词规则与职场礼貌缓冲}
丹麦语祈使句（用来提出请求、指令或建议）构造极其简单：\textbf{去掉动词原形末尾的 \textit{-e}}！
\begin{itemize}
  \item \textbf{标准去 \textit{-e}}：\textit{at vaske} $\rightarrow$ \dabold{Vask!} / \textit{at huske} $\rightarrow$ \dabold{Husk!} / \textit{at bruge} $\rightarrow$ \dabold{Brug!} / \textit{at tage} $\rightarrow$ \dabold{Tag!} / \textit{at være} $\rightarrow$ \dabold{Vær!}
  \item \textbf{无 \textit{-e} 词尾则不变}：\textit{at gå} $\rightarrow$ \da{Gå!} / \textit{at stå} $\rightarrow$ \da{Stå!}
  \item \textbf{双写辅音截尾规则（必记！）}：丹麦语词尾绝不保留两个相同的辅音字母！因此去掉 \textit{-e} 后，如果末尾剩下双写辅音，必须\textbf{砍掉一个}：\\
    \textit{at komme} $\rightarrow$ \dabold{Kom!}（不写 \textit{*Komm!}）/ \textit{at pakke} $\rightarrow$ \dabold{Pak!} / \textit{at slukke} $\rightarrow$ \dabold{Sluk!} / \textit{at lægge} $\rightarrow$ \dabold{Læg!} / \textit{at sætte} $\rightarrow$ \dabold{Sæt!}
  \item \textbf{礼貌加柔配方}：日常沟通中如果不想显得像发号施令，请在前面加上：\\
    \da{Vær sød at + 动词原形}（麻烦你/请你……）：\textit{Vær sød at hjælpe ved disken.}（如果对多人说，把 \textit{Vær} 后形容词变复数：\textit{Vær søde at...}）\\
    \da{Husk at + 动词原形}（记得……）：\textit{Husk at smile til kunderne!}
\end{itemize}
\end{grammatikbox}

\begin{grammatikbox}{语法点 2：泛指代词 \textit{man} 与职场规范三剑客}
英语用 \textit{you / people / one} 表示“人们通常/规定大家”，丹麦语最地道的词就是 \dabold{man}！注意：\textbf{\textit{man} 只能作主语}，而且绝不要和名词 \textit{en mand}（男人）混为一谈。搭配情态动词时，必须分清“禁止”和“不必”：

\vspace{0.3em}
\renewcommand{\arraystretch}{1.26}
{\small
\begin{tabularx}{\linewidth}{@{} >{\bfseries\raggedright\arraybackslash}p{3.8cm} >{\raggedright\arraybackslash}p{3.0cm} X @{}}
\toprule
\textbf{表达句型} & \textbf{准确中文含义} & \textbf{职场例句与易错警示} \\
\midrule
\da{man skal + 原形} & 大家必须 / 规定应当 & \textit{Man skal have rent tøj på.}（必须穿干净工作服。） \\
\midrule
\dabold{man må ikke + 原形} & \textbf{严禁 / 不准}（不是“不必”！） & \textit{Man må ikke bruge mobilen.}（绝对不准玩手机！） \\
\midrule
\textbf{\color{NCDTeal}man behøver ikke (at) + 原形} & \textbf{不必 / 不需要}（无强制义务） & \textit{Du behøver ikke at bage i dag.}（你今天不用烤面包。注意常保留 \textit{at}。） \\
\bottomrule
\end{tabularx}
}
\vspace{0.3em}

\textbf{高频地道搭配}：\da{skynde sig}（赶紧，快点——反身祈使：\textit{Skynd dig!}）/ \da{med det samme}（立刻，马上）/ \da{Helt i orden!}（没问题，好的！）。
\end{grammatikbox}

\begin{grammatikbox}{语音专题：喉塞音 \textit{stød}（\ipa{ˀ}）的基础感知}
很多同学觉得丹麦语听起来“一卡一卡的”，这主要是丹麦语标志性的\textbf{喉塞音（stød）}。它不是辅音，而是发长元音或浊辅音时，声带瞬间轻轻收紧造成的短促顿挫感（用 \ipa{ˀ} 标在音素右上角）。
\begin{enumerate}
  \item \textbf{与祈使句的奇妙联动}：双音节动词原形通常没有 \textit{stød}，一旦去掉 \textit{-e} 变成单音节祈使句，就会瞬间激活 \textit{stød}！\\
    \textit{at komme} \ipa{ˈkʰʌmə} $\rightarrow$ \dabold{Kom!} \ipa{ˈkʰʌmˀ} \quad|\quad \textit{at læse} \ipa{ˈlɛːsə} $\rightarrow$ \dabold{Læs!} \ipa{ˈlɛːˀs} \quad|\quad \textit{at spise} \ipa{ˈsbiːsə} $\rightarrow$ \dabold{Spis!} \ipa{ˈsbiːˀs}
  \item \textbf{靠 \textit{stød} 区分词义（经典最小对立对）}：\\
    \da{man} \ipa{man}（人们，无顿挫） vs \dabold{mand} \ipa{ˈmanˀ}（男人，有顿挫）\\
    \da{hun} \ipa{ˈhun}（她，无顿挫） vs \dabold{hund} \ipa{ˈhunˀ}（狗，有顿挫）\\
    \da{ven} \ipa{ˈvɛn}（朋友，无顿挫） vs \dabold{vand} \ipa{ˈvanˀ}（水）/ \dabold{vend!} \ipa{ˈvɛnˀ}（转过来！）
\end{enumerate}
\end{grammatikbox}

\ncdsubsec{五、Mønsterøvelser（新概念式句型操练）}

\textbf{练习 A：将下列动词改写为祈使句，并用 \textit{Vær sød at...} 改写成礼貌指令}
\begin{quote}
\small 例：\textit{at vaske hænder} $\rightarrow$ \textit{Vask hænder!} / \textit{Vær sød at vaske hænder!}
\end{quote}
\begin{enumerate}
  \item \textit{at pakke de friske rundstykker} $\rightarrow$ \underline{\hspace{5.5cm}}
  \item \textit{at slukke din mobiltelefon} $\rightarrow$ \underline{\hspace{5.5cm}}
  \item \textit{at lægge mobilen i skabet} $\rightarrow$ \underline{\hspace{5.5cm}}
  \item \textit{at komme ind i køkkenet} $\rightarrow$ \underline{\hspace{5.5cm}}
  \item \textit{at bruge rene handsker} $\rightarrow$ \underline{\hspace{5.5cm}}
\end{enumerate}

\textbf{练习 B：根据语境选填 \textit{man skal}、\textit{man må ikke} 或 \textit{man behøver ikke at}}
\begin{enumerate}
  \item I et bageri \underline{\hspace{3.2cm}} altid vaske hænderne, før man starter.
  \item Her bag disken \underline{\hspace{3.2cm}} spise eller drikke kaffe – det er forbudt!
  \item Som ny medarbejder \underline{\hspace{3.2cm}} bage brød den første dag; du skal bare hjælpe ved disken.
  \item Når man taler med kunderne, \underline{\hspace{3.2cm}} altid være venlig og smile.
  \item I vores kantine \underline{\hspace{3.2cm}} tage madpakke med, for der er billig frokost.
  \item På arbejdet \underline{\hspace{3.2cm}} bruge sin private mobiltelefon.
\end{enumerate}

\textbf{练习 C：中译丹实战}
\begin{enumerate}
  \item 请进厨房，穿上一件干净的围裙！
  \item 在面包店里，大家必须始终使用夹子或手套。
  \item 这里绝对不准戴戒指或使用手机。
  \item 你今天不需要烤面包，只需要帮忙照看柜台。
  \item 请关掉手机，记得微笑面对顾客！

\ncdsubsec{六、Resumé \& Mundtlig træning（口语复述与参考答案）}

\textbf{口语复述任务（对标 Modul 1--2 职场规范表达）}：请合上课文，用丹麦语回答 \textbf{Før du læser} 的两个问题，再用 \textbf{4--5 句话} 总结在这家面包店上班的“3 条必须/禁止规定”以及卢卡斯今天的工作任务。

\begin{answerbox}
\textbf{练习 A 答案}
\begin{enumerate}
  \item \da{Pak} de friske rundstykker! / \textit{Vær sød at pakke de friske rundstykker!}（双写 k 截掉一个！）
  \item \da{Sluk} din mobiltelefon! / \textit{Vær sød at slukke din mobiltelefon!}（双写 k 截掉一个！）
  \item \da{Læg} mobilen i skabet! / \textit{Vær sød at lægge mobilen i skabet!}（双写 g 截掉一个！）
  \item \da{Kom} ind i køkkenet! / \textit{Vær sød at komme ind i køkkenet!}（双写 m 截掉一个！）
  \item \da{Brug} rene handsker! / \textit{Vær sød at bruge rene handsker!}
\end{enumerate}

\textbf{练习 B 答案}
\begin{enumerate}
  \item \da{skal man}（句首地点状语触发 V2 倒装：\textit{I et bageri \dabold{skal man} altid...}！）
  \item \dabold{må man ikke}（严禁规定 + V2 倒装）
  \item \textbf{\color{NCDTeal}behøver man ikke at}（没有必要 + V2 倒装）
  \item \da{skal man}（从句在前触发主句 V2 倒装）
  \item \textbf{\color{NCDTeal}behøver man ikke at}（不必自己带午餐盒 + V2 倒装）
  \item \dabold{må man ikke}（禁止上班用私人手机 + V2 倒装）
\end{enumerate}

\textbf{练习 C 答案}
\begin{enumerate}
  \item \textit{Kom ind i køkkenet, og tag et rent forklæde på!}
  \item \textit{I et bageri skal man altid bruge en tang eller handsker.}
  \item \textit{Her må man ikke have ringe på eller bruge mobilen.}
  \item \textit{Du behøver ikke at bage brød i dag; du skal bare hjælpe ved disken.}
  \item \textit{Vær sød at slukke mobilen, og husk at smile til kunderne!}
\end{enumerate}

\textbf{Før du læser 参考回答 \& 口语复述范文（Resumé）}
\begin{itemize}
  \item \textbf{问题 1}：\textit{Man skal altid vaske hænderne grundigt først og have rent tøj på. Man må ikke have ringe eller ur på, og man må heller ikke bruge sin mobil ved disken.}
  \item \textbf{问题 2}：\textit{Nej, Lukas behøver ikke at bage brødet i dag. Han skal hjælpe ved disken og pakke de friske rundstykker.}
  \item \textbf{口语复述范文}：\textit{Det er Lukas' første dag i bageriet. Chefen Mette fortæller ham om reglerne: man skal altid vaske hænder først og bruge tang eller handsker, og man må ikke have ringe, ur eller mobiltelefon ved disken. Lukas behøver ikke at bage brød endnu. Han skal bare tage et rent forklæde på, hjælpe ved disken og huske at smile til kunderne.}
\end{itemize}
\end{answerbox}
